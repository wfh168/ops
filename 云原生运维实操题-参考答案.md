# 云原生运维实操题 - 参考答案

## 第一阶段：容器化改造

### 1.1 前端 Dockerfile

```dockerfile
# 构建阶段
FROM node:16-alpine AS builder
WORKDIR /app
COPY package*.json ./
RUN npm ci --only=production
COPY . .
RUN npm run build

# 运行阶段
FROM nginx:alpine
# 创建非 root 用户
RUN addgroup -g 1001 -S nginx-user && \
    adduser -u 1001 -S nginx-user -G nginx-user && \
    chown -R nginx-user:nginx-user /var/cache/nginx && \
    chown -R nginx-user:nginx-user /var/log/nginx && \
    touch /var/run/nginx.pid && \
    chown -R nginx-user:nginx-user /var/run/nginx.pid

# 设置时区
RUN apk add --no-cache tzdata && \
    cp /usr/share/zoneinfo/Asia/Shanghai /etc/localtime && \
    echo "Asia/Shanghai" > /etc/timezone && \
    apk del tzdata

# 复制构建产物和配置
COPY --from=builder /app/dist /usr/share/nginx/html
COPY nginx.conf /etc/nginx/nginx.conf

# 切换用户
USER nginx-user

# 健康检查
HEALTHCHECK --interval=30s --timeout=3s --start-period=5s --retries=3 \
    CMD wget --quiet --tries=1 --spider http://localhost:80/health || exit 1

EXPOSE 80
CMD ["nginx", "-g", "daemon off;"]
```

### 1.2 后端 Dockerfile

```dockerfile
FROM openjdk:11-jre-slim

# 创建非 root 用户
RUN groupadd -r appuser && useradd -r -g appuser appuser

# 设置时区
RUN apt-get update && \
    apt-get install -y --no-install-recommends tzdata && \
    ln -sf /usr/share/zoneinfo/Asia/Shanghai /etc/localtime && \
    echo "Asia/Shanghai" > /etc/timezone && \
    apt-get clean && \
    rm -rf /var/lib/apt/lists/*

WORKDIR /app

# 复制应用
COPY --chown=appuser:appuser app.jar /app/app.jar

# 切换用户
USER appuser

# 健康检查
HEALTHCHECK --interval=30s --timeout=3s --start-period=40s --retries=3 \
    CMD curl -f http://localhost:8080/actuator/health || exit 1

EXPOSE 8080

# JVM 优化参数
ENV JAVA_OPTS="-Xms512m -Xmx1024m -XX:+UseG1GC"

ENTRYPOINT ["sh", "-c", "java $JAVA_OPTS -jar /app/app.jar"]
```

  name: nginx-config
  namespace: ecommerce-prod

  nginx.conf: |
    user nginx;
    worker_processes auto;
    error_log /var/log/nginx/error.log warn;
    pid /var/run/nginx.pid;
    
    events {
        worker_connections 1024;
    }
    
    http {
        include /etc/nginx/mime.types;
        default_type application/octet-stream;
        
        log_format main '$remote_addr - $remote_user [$time_local] "$request" '
                        '$status $body_bytes_sent "$http_referer" '
                        '"$http_user_agent" "$http_x_forwarded_for"';
        
        access_log /var/log/nginx/access.log main;
        
        sendfile on;
        keepalive_timeout 65;
        
        upstream backend {
            server backend-service:8080;
        }
        
        server {
            listen 80;
            server_name _;
            
            location / {
                root /usr/share/nginx/html;
                index index.html;
                try_files $uri $uri/ /index.html;
            }
            
            location /api/ {
                proxy_pass http://backend/;
                proxy_set_header Host $host;
                proxy_set_header X-Real-IP $remote_addr;
                proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            }
            
            location /health {
                access_log off;
                return 200 "healthy\n";
            }
        }
    }

### 3.2 Prometheus 告警规则

```yaml
# alert-rules.yaml
apiVersion: monitoring.coreos.com/v1
kind: PrometheusRule
metadata:
  name: ecommerce-alerts
  namespace: ecommerce-prod
spec:
  groups:
  - name: pod-alerts
    interval: 30s
    rules:
    - alert: PodRestartingTooOften
      expr: rate(kube_pod_container_status_restarts_total{namespace="ecommerce-prod"}[1h]) > 5
      for: 5m
      labels:
        severity: warning
      annotations:
        summary: "Pod {{ $labels.pod }} 重启过于频繁"
        description: "Pod {{ $labels.pod }} 在过去1小时内重启了 {{ $value }} 次"
    
    - alert: PodCPUUsageHigh
      expr: sum(rate(container_cpu_usage_seconds_total{namespace="ecommerce-prod"}[5m])) by (pod) > 0.8
      for: 5m
      labels:
        severity: warning
      annotations:
        summary: "Pod {{ $labels.pod }} CPU使用率过高"
        description: "Pod {{ $labels.pod }} CPU使用率为 {{ $value | humanizePercentage }}"
    
    - alert: PodMemoryUsageHigh
      expr: sum(container_memory_working_set_bytes{namespace="ecommerce-prod"}) by (pod) / sum(container_spec_memory_limit_bytes{namespace="ecommerce-prod"}) by (pod) > 0.85
      for: 5m
      labels:
        severity: warning
      annotations:
        summary: "Pod {{ $labels.pod }} 内存使用率过高"
        description: "Pod {{ $labels.pod }} 内存使用率为 {{ $value | humanizePercentage }}"
  
  - name: application-alerts
    interval: 30s
    rules:
    - alert: HighResponseTime
      expr: histogram_quantile(0.95, sum(rate(http_server_requests_seconds_bucket{namespace="ecommerce-prod"}[5m])) by (le, uri)) > 2
      for: 5m
      labels:
        severity: warning
      annotations:
        summary: "接口 {{ $labels.uri }} 响应时间过长"
        description: "接口 {{ $labels.uri }} P95响应时间为 {{ $value }}s"
    
    - alert: HighErrorRate
      expr: sum(rate(http_server_requests_seconds_count{namespace="ecommerce-prod",status=~"5.."}[5m])) / sum(rate(http_server_requests_seconds_count{namespace="ecommerce-prod"}[5m])) > 0.05
      for: 5m
      labels:
        severity: critical
      annotations:
        summary: "应用错误率过高"
        description: "应用错误率为 {{ $value | humanizePercentage }}"
  
  - name: database-alerts
    interval: 30s
    rules:
    - alert: MySQLDown
      expr: mysql_up{namespace="ecommerce-prod"} == 0
      for: 1m
      labels:
        severity: critical
      annotations:
        summary: "MySQL 数据库不可用"
        description: "MySQL 实例 {{ $labels.instance }} 已宕机"
    
    - alert: MySQLSlowQueries
      expr: rate(mysql_global_status_slow_queries{namespace="ecommerce-prod"}[5m]) > 10
      for: 5m
      labels:
        severity: warning
      annotations:
        summary: "MySQL 慢查询过多"
        description: "MySQL 慢查询速率为 {{ $value }}/s"
```

### 4.3 健康检查脚本

```bash
#!/bin/bash
# health-check.sh

NAMESPACE="ecommerce-prod"
MAX_RETRIES=30
RETRY_INTERVAL=10

check_service() {
    local service=$1
    local path=$2
    local expected_code=$3
    
    echo "检查 ${service}..."
    
    for i in $(seq 1 ${MAX_RETRIES}); do
        POD=$(kubectl get pod -n ${NAMESPACE} -l app=${service} -o jsonpath='{.items[0].metadata.name}')
        
        if [ -z "$POD" ]; then
            echo "  ❌ 未找到 ${service} Pod"
            return 1
        fi
        
        STATUS=$(kubectl exec -n ${NAMESPACE} ${POD} -- curl -s -o /dev/null -w "%{http_code}" http://localhost:${path} 2>/dev/null || echo "000")
        
        if [ "$STATUS" = "$expected_code" ]; then
            echo "  ✅ ${service} 健康检查通过"
            return 0
        fi
        
        echo "  ⏳ 等待 ${service} 就绪... (${i}/${MAX_RETRIES})"
        sleep ${RETRY_INTERVAL}
    done
    
    echo "  ❌ ${service} 健康检查失败"
    return 1
}

echo "🏥 开始健康检查..."

# 检查后端服务
check_service "backend" "8080/actuator/health" "200"
BACKEND_STATUS=$?

# 检查前端服务
check_service "frontend" "80/health" "200"
FRONTEND_STATUS=$?

# 检查数据库连接
echo "检查数据库连接..."
MYSQL_POD=$(kubectl get pod -n ${NAMESPACE} -l app=mysql -o jsonpath='{.items[0].metadata.name}')
kubectl exec -n ${NAMESPACE} ${MYSQL_POD} -- mysql -u root -p${MYSQL_ROOT_PASSWORD} -e "SELECT 1" > /dev/null 2>&1
MYSQL_STATUS=$?

if [ $MYSQL_STATUS -eq 0 ]; then
    echo "  ✅ MySQL 连接正常"
else
    echo "  ❌ MySQL 连接失败"
fi

# 总结
if [ $BACKEND_STATUS -eq 0 ] && [ $FRONTEND_STATUS -eq 0 ] && [ $MYSQL_STATUS -eq 0 ]; then
    echo "✅ 所有健康检查通过！"
    exit 0
else
    echo "❌ 健康检查失败，请检查日志"
    exit 1
fi
```

### 4.4 回滚脚本

#!/bin/bash
# rollback.sh

NAMESPACE="ecommerce-prod"
