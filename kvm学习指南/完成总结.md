## 💡 实用技巧

### 命令速查

```bash
# 虚拟机管理
virsh list --all              # 列出所有虚拟机
virsh start vm-name           # 启动虚拟机
virsh shutdown vm-name        # 关闭虚拟机
virsh dominfo vm-name         # 查看虚拟机信息

# 网络管理
virsh net-list --all          # 列出所有网络
virsh net-info default        # 查看网络信息
virsh domifaddr vm-name       # 查看虚拟机 IP

# 存储管理
virsh pool-list --all         # 列出所有存储池
virsh vol-list default        # 列出存储卷
virsh domblklist vm-name      # 查看虚拟机磁盘
```

### 快捷脚本

```bash
# 批量启动虚拟机
for vm in $(virsh list --all --name); do
    virsh start $vm
done

# 批量关闭虚拟机
for vm in $(virsh list --name); do
    virsh shutdown $vm
done

# 查看所有虚拟机状态
virsh list --all
```
