# 流程控制与函数

## 一、条件判断

### 1.1 if 语句

#### 基本语法

```bash
# 单分支
if [ condition ]; then
    commands
fi

# 双分支
if [ condition ]; then
    commands1
else
    commands2
fi

# 多分支
if [ condition1 ]; then
    commands1
elif [ condition2 ]; then
    commands2
else
    commands3
fi
```

#### 实例1：检查文件

```bash
#!/bin/bash

file="/etc/passwd"

if [ -f "$file" ]; then
    echo "$file 是一个普通文件"
    echo "文件大小：$(du -h $file | cut -f1)"
    echo "文件权限：$(stat -c %a $file)"
else
    echo "$file 不是普通文件"
fi
```

#### 实例2：比较数值

```bash
#!/bin/bash

read -p "请输入你的分数（0-100）：" score

if [ $score -ge 90 ]; then
    echo "优秀"
elif [ $score -ge 80 ]; then
    echo "良好"
elif [ $score -ge 60 ]; then
    echo "及格"
else
    echo "不及格"
fi
```

#### 实例3：检查服务状态

```bash
#!/bin/bash

service="nginx"

if systemctl is-active --quiet $service; then
    echo "$service 正在运行"
    echo "PID: $(systemctl show -p MainPID $service | cut -d= -f2)"
else
    echo "$service 未运行"
    read -p "是否启动服务？(y/n) " answer
    if [ "$answer" = "y" ]; then
        systemctl start $service
        echo "$service 已启动"
    fi
fi
```

### 1.2 case 语句

#### 基本语法

```bash
case $variable in
    pattern1)
        commands1
        ;;

        commands2
        ;;
    pattern3|pattern4)
        commands3
        ;;
    *)
        default_commands
        ;;
esac
```

#### 实例2：服务管理

```bash
#!/bin/bash

if [ $# -ne 2 ]; then
    echo "用法：$0 <服务名> <操作>"

        ;;
```

```bash
        echo "未知文件类型"
        ;;
esac
```

## 二、循环结构

### 2.1 for 循环

#### 语法1：列表遍历

```bash
#!/bin/bash

# 遍历列表
for item in item1 item2 item3; do
    echo $item
done

# 遍历数组
fruits=("苹果" "香蕉" "橙子")
for fruit in "${fruits[@]}"; do
    echo "水果：$fruit"
done

# 遍历命令输出
for file in $(ls *.txt); do
    echo "文件：$file"
done

# 遍历文件内容
for line in $(cat file.txt); do
    echo $line
done
```

#### 语法2：C 风格

```bash
#!/bin/bash

# 基本用法
for ((i=1; i<=10; i++)); do
    echo "数字：$i"
done

# 步长为2
for ((i=0; i<=20; i+=2)); do
    echo $i
done

# 倒序
for ((i=10; i>=1; i--)); do
    echo $i
done
```

#### 实例1：批量重命名

```bash
#!/bin/bash

# 将所有 .txt 文件重命名为 .bak
for file in *.txt; do
    if [ -f "$file" ]; then
        newname="${file%.txt}.bak"
        mv "$file" "$newname"
        echo "重命名：$file -> $newname"
    fi
done
```

#### 实例2：批量创建用户

```bash
#!/bin/bash

# 从文件读取用户列表并创建
while IFS=: read -r username password; do
    if id "$username" &>/dev/null; then
        echo "用户 $username 已存在"
    else
        useradd "$username"
        echo "$username:$password" | chpasswd
        echo "创建用户：$username"
    fi
done < users.txt
```

#### 实例3：批量 ping 测试

```bash
#!/bin/bash

network="192.168.1"

echo "开始 ping 测试..."
for i in {1..254}; do
    ip="$network.$i"
    if ping -c 1 -W 1 $ip &>/dev/null; then
        echo "$ip 在线"
    fi
done
```

### 2.2 while 循环

#### 基本语法

```bash
#!/bin/bash

# 基本用法
count=1
while [ $count -le 5 ]; do
    echo "计数：$count"
    ((count++))
done

# 无限循环
while true; do
    echo "按 Ctrl+C 退出"
    sleep 1
done

# 读取文件
while read line; do
    echo "行内容：$line"
done < file.txt
```

#### 实例1：监控进程

```bash
#!/bin/bash

process="nginx"

while true; do
    if pgrep $process > /dev/null; then
        echo "$(date): $process 正在运行"
    else
        echo "$(date): $process 已停止，正在重启..."
        systemctl start $process
    fi
    sleep 60
done
```

#### 实例2：等待服务启动

```bash
#!/bin/bash

service="mysql"
max_wait=30
count=0

echo "等待 $service 启动..."

while ! systemctl is-active --quiet $service; do
    if [ $count -ge $max_wait ]; then
        echo "超时：$service 未能在 $max_wait 秒内启动"
        exit 1
    fi
    echo "等待中... ($count/$max_wait)"
    sleep 1
    ((count++))
done

echo "$service 已启动"
```

### 2.4 循环控制

```bash
#!/bin/bash

# break：跳出循环
for i in {1..10}; do
    if [ $i -eq 5 ]; then
        break
    fi
    echo $i
done

# continue：跳过本次循环
for i in {1..10}; do
    if [ $i -eq 5 ]; then
        continue
    fi
    echo $i
done

# 嵌套循环的 break
for i in {1..3}; do
    for j in {1..3}; do
        if [ $j -eq 2 ]; then
            break 2  # 跳出两层循环
        fi
        echo "$i-$j"
    done
done
```

### 3.2 函数参数

```bash
    echo "第二个参数：$2"
```

### 3.3 函数返回值

```bash
#!/bin/bash

# 方法1：使用 return（只能返回 0-255 的整数）
function add() {
    local result=$(($1 + $2))
    return $result
}

add 10 20
echo "返回值：$?"

# 方法2：使用 echo（推荐）
function multiply() {
    local result=$(($1 * $2))
    echo $result
}

result=$(multiply 10 20)
echo "结果：$result"

# 方法3：使用全局变量
function divide() {
    RESULT=$(echo "scale=2; $1 / $2" | bc)
}

divide 10 3
echo "结果：$RESULT"
```

### 3.4 局部变量和全局变量

```bash
#!/bin/bash

# 全局变量
global_var="全局变量"

function test_scope() {
    # 局部变量
    local local_var="局部变量"
    
    # 修改全局变量
    global_var="修改后的全局变量"
    
    echo "函数内："
    echo "  局部变量：$local_var"
    echo "  全局变量：$global_var"
}

test_scope

echo "函数外："
echo "  局部变量：$local_var"  # 无法访问
echo "  全局变量：$global_var"
```

### 3.5 函数库

创建函数库文件 `lib.sh`：
```bash
#!/bin/bash

# 日志函数
function log_info() {
    echo "[$(date '+%Y-%m-%d %H:%M:%S')] [INFO] $1"
}

function log_error() {
    echo "[$(date '+%Y-%m-%d %H:%M:%S')] [ERROR] $1" >&2
}

# 检查命令是否存在
function command_exists() {
    command -v "$1" &>/dev/null
}

# 检查是否为 root 用户
function check_root() {
    if [ $EUID -ne 0 ]; then
        log_error "此脚本需要 root 权限"
        exit 1
    fi
}

# 确认操作
function confirm() {
    read -p "$1 (y/n): " answer
    case $answer in
        [Yy]* ) return 0;;
        * ) return 1;;
    esac
}
```

使用函数库：
```bash
#!/bin/bash

# 导入函数库
source ./lib.sh

# 使用函数
log_info "脚本开始执行"

if command_exists nginx; then
    log_info "Nginx 已安装"
else
    log_error "Nginx 未安装"
fi

if confirm "是否继续？"; then
    log_info "用户选择继续"
else
    log_info "用户取消操作"
    exit 0
fi
```

## 四、实战案例

### 案例1：系统监控脚本

```bash
#!/bin/bash

# 监控阈值
CPU_THRESHOLD=80
MEM_THRESHOLD=80
DISK_THRESHOLD=80

# 日志函数
log() {
    echo "[$(date '+%Y-%m-%d %H:%M:%S')] $1" | tee -a /var/log/monitor.log
}

# 检查 CPU 使用率
check_cpu() {
    cpu_usage=$(top -bn1 | grep "Cpu(s)" | awk '{print $2}' | cut -d'%' -f1)
    cpu_usage=${cpu_usage%.*}
    
    if [ $cpu_usage -gt $CPU_THRESHOLD ]; then
        log "警告：CPU 使用率过高 ($cpu_usage%)"
        return 1
    else
        log "正常：CPU 使用率 $cpu_usage%"
        return 0
    fi
}

# 检查内存使用率
check_memory() {
    mem_usage=$(free | grep Mem | awk '{printf "%.0f", $3/$2 * 100}')
    
    if [ $mem_usage -gt $MEM_THRESHOLD ]; then
        log "警告：内存使用率过高 ($mem_usage%)"
        return 1
    else
        log "正常：内存使用率 $mem_usage%"
        return 0
    fi
}

# 检查磁盘使用率
check_disk() {
    while read line; do
        usage=$(echo $line | awk '{print $5}' | cut -d'%' -f1)
        mount=$(echo $line | awk '{print $6}')
        
        if [ $usage -gt $DISK_THRESHOLD ]; then
            log "警告：磁盘 $mount 使用率过高 ($usage%)"
        fi
    done < <(df -h | grep -E '^/dev/')
}

# 主函数
main() {
    log "========== 开始系统监控 =========="
    
    check_cpu
    check_memory
    check_disk
    
    log "========== 监控完成 =========="
}

main
```

### 案例2：自动备份脚本

```bash
#!/bin/bash

# 配置
SOURCE_DIR="/var/www/html"
BACKUP_DIR="/backup"
DATE=$(date +%Y%m%d_%H%M%S)
BACKUP_FILE="backup_$DATE.tar.gz"
KEEP_DAYS=7

# 日志函数
log() {
    echo "[$(date '+%Y-%m-%d %H:%M:%S')] $1"
}

# 创建备份
create_backup() {
    log "开始备份 $SOURCE_DIR"
    
    if [ ! -d "$BACKUP_DIR" ]; then
        mkdir -p "$BACKUP_DIR"
    fi
    
    tar -czf "$BACKUP_DIR/$BACKUP_FILE" "$SOURCE_DIR" 2>/dev/null
    
    if [ $? -eq 0 ]; then
        log "备份成功：$BACKUP_FILE"
        log "备份大小：$(du -h $BACKUP_DIR/$BACKUP_FILE | cut -f1)"
        return 0
    else
        log "备份失败"
        return 1
    fi
}

# 清理旧备份
cleanup_old_backups() {
    log "清理 $KEEP_DAYS 天前的备份"
    
    find "$BACKUP_DIR" -name "backup_*.tar.gz" -mtime +$KEEP_DAYS -delete
    
    log "当前备份文件："
    ls -lh "$BACKUP_DIR"/backup_*.tar.gz 2>/dev/null || log "无备份文件"
}

# 主函数
main() {
    log "========== 开始备份任务 =========="
    
    if create_backup; then
        cleanup_old_backups
        log "备份任务完成"
    else
        log "备份任务失败"
        exit 1
    fi
}

main
```

### 案例3：批量主机管理

```bash
#!/bin/bash

# 主机列表文件
HOSTS_FILE="hosts.txt"

# 执行远程命令
execute_remote() {
    local host=$1
    local command=$2
    
    echo "========== $host =========="
    ssh -o ConnectTimeout=5 root@$host "$command" 2>/dev/null
    
    if [ $? -eq 0 ]; then
        echo "✓ 执行成功"
    else
        echo "✗ 执行失败"
    fi
    echo ""
}

# 批量执行
batch_execute() {
    local command=$1
    
    if [ ! -f "$HOSTS_FILE" ]; then
        echo "错误：主机列表文件不存在"
        exit 1
    fi
    
    while read host; do
        # 跳过空行和注释
        [[ -z "$host" || "$host" =~ ^# ]] && continue
        
        execute_remote "$host" "$command"
    done < "$HOSTS_FILE"
}

# 主函数
main() {
    if [ $# -eq 0 ]; then
        echo "用法：$0 <命令>"
        echo "示例：$0 'uptime'"
        exit 1
    fi
    
    command="$*"
    echo "执行命令：$command"
    echo ""
    
    batch_execute "$command"
}

main "$@"
```
