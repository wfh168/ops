# 监控、优化与自动化

## 一、JVM 监控

### 1.1 JVM 监控工具

#### jps - 查看 Java 进程

```bash
# 查看所有 Java 进程
jps -l

# 查看进程和主类
jps -lv

# 输出示例
12345 com.example.MyApp -Xms2g -Xmx2g
```

#### jstat - JVM 统计信息

```bash
# 查看 GC 统计
jstat -gc 12345 1000 10

# 查看 GC 汇总
jstat -gcutil 12345 1000

# 查看类加载统计
jstat -class 12345

# 查看编译统计
jstat -compiler 12345

# 输出说明
S0C    S1C    S0U    S1U      EC       EU        OC         OU       MC     MU    CCSC   CCSU   YGC     YGCT    FGC    FGCT     GCT
10240  10240  0      9216     81920    40960     204800     102400   51200  48000  6144   5800   100     0.500   5      0.200    0.700
```

#### jmap - 内存映射

```bash
# 查看堆内存使用
jmap -heap 12345

# 查看对象统计
jmap -histo 12345 | head -20

# 生成堆转储文件
jmap -dump:format=b,file=heapdump.hprof 12345

# 查看类加载器统计
jmap -clstats 12345
```

#### jstack - 线程堆栈

```bash
# 查看线程堆栈
jstack 12345

# 查看死锁
jstack -l 12345 | grep -A 10 "deadlock"

# 输出到文件
jstack 12345 > thread_dump.txt
```

#### jcmd - 综合诊断工具

```bash
# 查看可用命令
jcmd 12345 help

# 查看 JVM 信息
jcmd 12345 VM.version
jcmd 12345 VM.flags
jcmd 12345 VM.system_properties

# 生成堆转储
jcmd 12345 GC.heap_dump /tmp/heapdump.hprof

# 查看 GC 信息
jcmd 12345 GC.class_histogram
jcmd 12345 GC.heap_info

# 线程信息
jcmd 12345 Thread.print
```

## 三、日志管理

### 3.1 Logback 配置

```xml
<!-- logback-spring.xml -->
<?xml version="1.0" encoding="UTF-8"?>
<configuration>
    
    <!-- 日志文件路径 -->
    <property name="LOG_PATH" value="/var/log/myapp"/>
    <property name="LOG_FILE" value="${LOG_PATH}/app.log"/>
    
    <!-- 控制台输出 -->
    <appender name="CONSOLE" class="ch.qos.logback.core.ConsoleAppender">
        <encoder>
            <pattern>%d{yyyy-MM-dd HH:mm:ss.SSS} [%thread] %-5level %logger{36} - %msg%n</pattern>
            <charset>UTF-8</charset>
        </encoder>
    </appender>
    
    <!-- 文件输出 -->
    <appender name="FILE" class="ch.qos.logback.core.rolling.RollingFileAppender">
        <file>${LOG_FILE}</file>
        <encoder>
            <pattern>%d{yyyy-MM-dd HH:mm:ss.SSS} [%thread] %-5level %logger{36} - %msg%n</pattern>
            <charset>UTF-8</charset>
        </encoder>
        
        <!-- 滚动策略 -->
        <rollingPolicy class="ch.qos.logback.core.rolling.SizeAndTimeBasedRollingPolicy">
            <fileNamePattern>${LOG_PATH}/app-%d{yyyy-MM-dd}.%i.log.gz</fileNamePattern>
            <maxFileSize>100MB</maxFileSize>
            <maxHistory>30</maxHistory>
            <totalSizeCap>10GB</totalSizeCap>
        </rollingPolicy>
    </appender>
    
    <!-- 错误日志 -->
    <appender name="ERROR_FILE" class="ch.qos.logback.core.rolling.RollingFileAppender">
        <file>${LOG_PATH}/error.log</file>
        <filter class="ch.qos.logback.classic.filter.LevelFilter">
            <level>ERROR</level>
            <onMatch>ACCEPT</onMatch>
            <onMismatch>DENY</onMismatch>
        </filter>
        <encoder>
            <pattern>%d{yyyy-MM-dd HH:mm:ss.SSS} [%thread] %-5level %logger{36} - %msg%n</pattern>
        </encoder>
        <rollingPolicy class="ch.qos.logback.core.rolling.SizeAndTimeBasedRollingPolicy">
            <fileNamePattern>${LOG_PATH}/error-%d{yyyy-MM-dd}.%i.log.gz</fileNamePattern>
            <maxFileSize>100MB</maxFileSize>
            <maxHistory>30</maxHistory>
        </rollingPolicy>
    </appender>
    
    <!-- 异步日志 -->
    <appender name="ASYNC_FILE" class="ch.qos.logback.classic.AsyncAppender">
        <appender-ref ref="FILE"/>
        <queueSize>512</queueSize>
        <discardingThreshold>0</discardingThreshold>
    </appender>
    
    <!-- 日志级别 -->
    <logger name="com.example.myapp" level="INFO"/>
    <logger name="org.springframework" level="WARN"/>
    <logger name="org.hibernate" level="WARN"/>
    
    <root level="INFO">
        <appender-ref ref="CONSOLE"/>
        <appender-ref ref="ASYNC_FILE"/>
        <appender-ref ref="ERROR_FILE"/>
    </root>
    
</configuration>
```

## 四、性能优化

### 4.1 JVM 调优

#### G1 垃圾回收器（推荐）

```bash
java \
  # 堆内存
  -Xms4g \
  -Xmx4g \
  -Xmn2g \
  
  # 元空间
  -XX:MetaspaceSize=256m \
  -XX:MaxMetaspaceSize=512m \
  
  # G1 垃圾回收器
  -XX:+UseG1GC \
  -XX:MaxGCPauseMillis=200 \
  -XX:G1HeapRegionSize=16m \
  -XX:G1ReservePercent=10 \
  -XX:InitiatingHeapOccupancyPercent=45 \
  
  # GC 日志
  -Xlog:gc*:file=/var/log/myapp/gc.log:time,uptime:filecount=10,filesize=100m \
  
  # OOM 处理
  -XX:+HeapDumpOnOutOfMemoryError \
  -XX:HeapDumpPath=/var/log/myapp/heapdump.hprof \
  
  -jar myapp.jar
```

#### ZGC 垃圾回收器（低延迟）

```bash
java \
  -Xms8g \
  -Xmx8g \
  -XX:+UseZGC \
  -XX:ZCollectionInterval=120 \
  -XX:ZAllocationSpikeTolerance=5 \
  -Xlog:gc*:file=/var/log/myapp/gc.log:time,uptime:filecount=10,filesize=100m \
  -jar myapp.jar
```

```java
// 批量操作
@Transactional
public void batchInsert(List<User> users) {
    int batchSize = 100;
    for (int i = 0; i < users.size(); i++) {
        entityManager.persist(users.get(i));
        if (i % batchSize == 0 && i > 0) {
            entityManager.flush();
            entityManager.clear();
```

```java
@Service
public class UserService {
    
    // 方法级缓存
    @Cacheable(value = "users", key = "#id")

        return userRepository.findById(id).orElse(null);

    @CacheEvict(value = "users", key = "#id")

    // 清空缓存
    @CacheEvict(value = "users", allEntries = true)

        // 清空所有缓存
    }
}
```

## 五、自动化部署

### 5.1 部署脚本

```bash
#!/bin/bash
# deploy.sh

set -e

APP_NAME="myapp"
APP_VERSION="1.0.0"
APP_JAR="${APP_NAME}-${APP_VERSION}.jar"
APP_DIR="/opt/${APP_NAME}"
BACKUP_DIR="/opt/${APP_NAME}/backup"
LOG_DIR="/var/log/${APP_NAME}"

echo "=== 开始部署 ${APP_NAME} ==="

# 1. 备份当前版本
if [ -f "${APP_DIR}/${APP_JAR}" ]; then
    echo "备份当前版本..."
    mkdir -p ${BACKUP_DIR}
    cp ${APP_DIR}/${APP_JAR} ${BACKUP_DIR}/${APP_JAR}.$(date +%Y%m%d%H%M%S)
fi

# 2. 停止应用
echo "停止应用..."
systemctl stop ${APP_NAME} || true

# 3. 部署新版本
echo "部署新版本..."
cp target/${APP_JAR} ${APP_DIR}/

# 4. 启动应用
echo "启动应用..."
systemctl start ${APP_NAME}

# 5. 健康检查
echo "健康检查..."
for i in {1..30}; do
    if curl -f http://localhost:8080/actuator/health > /dev/null 2>&1; then
        echo "应用启动成功！"
        exit 0
    fi
    echo "等待应用启动... ($i/30)"
    sleep 2
done

# 6. 启动失败，回滚
echo "应用启动失败，开始回滚..."
systemctl stop ${APP_NAME}
LAST_BACKUP=$(ls -t ${BACKUP_DIR}/*.jar | head -1)
if [ -n "$LAST_BACKUP" ]; then
    cp $LAST_BACKUP ${APP_DIR}/${APP_JAR}
    systemctl start ${APP_NAME}
    echo "已回滚到上一个版本"
fi

exit 1
```

### 5.2 蓝绿部署

```bash
#!/bin/bash
# blue-green-deploy.sh

set -e

APP_NAME="myapp"
BLUE_PORT=8080
GREEN_PORT=8081
NGINX_UPSTREAM="/etc/nginx/conf.d/upstream.conf"

# 检测当前活跃环境
CURRENT_PORT=$(curl -s http://localhost/actuator/info | jq -r '.port' || echo $BLUE_PORT)

if [ "$CURRENT_PORT" == "$BLUE_PORT" ]; then
    ACTIVE_ENV="blue"
    STANDBY_ENV="green"
    STANDBY_PORT=$GREEN_PORT
else
    ACTIVE_ENV="green"
    STANDBY_ENV="blue"
    STANDBY_PORT=$BLUE_PORT
fi

echo "当前活跃环境: $ACTIVE_ENV"
echo "待部署环境: $STANDBY_ENV"

# 1. 部署到待机环境
echo "部署到 $STANDBY_ENV 环境..."
java -jar myapp.jar --server.port=$STANDBY_PORT &
STANDBY_PID=$!

# 2. 健康检查
echo "健康检查..."
for i in {1..30}; do
    if curl -f http://localhost:$STANDBY_PORT/actuator/health > /dev/null 2>&1; then
        echo "$STANDBY_ENV 环境启动成功！"
        break
    fi
    if [ $i -eq 30 ]; then
        echo "$STANDBY_ENV 环境启动失败！"
        kill $STANDBY_PID
        exit 1
    fi
    sleep 2
done

# 3. 切换流量
echo "切换流量到 $STANDBY_ENV 环境..."
cat > $NGINX_UPSTREAM <<EOF
upstream backend {
    server 127.0.0.1:$STANDBY_PORT;
}
EOF

nginx -s reload

# 4. 停止旧环境
echo "停止 $ACTIVE_ENV 环境..."
sleep 5
pkill -f "server.port=$CURRENT_PORT"

echo "部署完成！"
```

### 5.3 灰度发布

```bash
#!/bin/bash
# canary-deploy.sh

set -e

APP_NAME="myapp"
OLD_VERSION="1.0.0"
NEW_VERSION="1.1.0"
CANARY_WEIGHT=10  # 10% 流量到新版本

echo "=== 灰度发布 ${APP_NAME} ${NEW_VERSION} ==="

# 1. 部署新版本（单个实例）
echo "部署新版本实例..."
java -jar ${APP_NAME}-${NEW_VERSION}.jar --server.port=8082 &
NEW_PID=$!

# 2. 健康检查
echo "健康检查..."
for i in {1..30}; do
    if curl -f http://localhost:8082/actuator/health > /dev/null 2>&1; then
        echo "新版本启动成功！"
        break
    fi
    if [ $i -eq 30 ]; then
        echo "新版本启动失败！"
        kill $NEW_PID
        exit 1
    fi
    sleep 2
done

# 3. 配置灰度流量
echo "配置灰度流量（${CANARY_WEIGHT}%）..."
cat > /etc/nginx/conf.d/upstream.conf <<EOF
upstream backend {
    server 127.0.0.1:8080 weight=90;  # 旧版本
    server 127.0.0.1:8081 weight=90;  # 旧版本
    server 127.0.0.1:8082 weight=${CANARY_WEIGHT};  # 新版本
}
EOF

nginx -s reload

# 4. 监控新版本
echo "监控新版本（5分钟）..."
sleep 300

# 5. 检查错误率
ERROR_RATE=$(curl -s http://localhost:8082/actuator/metrics/http.server.requests | jq '.measurements[0].value')
if (( $(echo "$ERROR_RATE > 0.01" | bc -l) )); then
    echo "错误率过高，回滚！"
    kill $NEW_PID
    # 恢复 Nginx 配置
    cat > /etc/nginx/conf.d/upstream.conf <<EOF
upstream backend {
    server 127.0.0.1:8080;
    server 127.0.0.1:8081;
}
EOF
    nginx -s reload
    exit 1
fi

# 6. 逐步增加流量
echo "逐步增加流量..."
for weight in 20 50 100; do
    echo "调整流量到 ${weight}%..."
    cat > /etc/nginx/conf.d/upstream.conf <<EOF
upstream backend {
    server 127.0.0.1:8080 weight=$((100-weight));
    server 127.0.0.1:8081 weight=$((100-weight));
    server 127.0.0.1:8082 weight=${weight};
}
EOF
    nginx -s reload
    sleep 300
done

# 7. 完全切换到新版本
echo "完全切换到新版本..."
pkill -f "${APP_NAME}-${OLD_VERSION}.jar"

echo "灰度发布完成！"
```
