# 函数与运算符

## 一、字符串函数

### 1.1 字符串拼接

```sql
-- CONCAT：拼接字符串
SELECT CONCAT('Hello', ' ', 'World');  -- Hello World
SELECT CONCAT(name, '-', age) FROM students;

-- CONCAT_WS：使用分隔符拼接
SELECT CONCAT_WS('-', 'A', 'B', 'C');  -- A-B-C
SELECT CONCAT_WS(',', name, age, gender) FROM students;
```

### 1.2 字符串长度

```sql
-- LENGTH：字节长度
SELECT LENGTH('Hello');  -- 5
SELECT LENGTH('你好');   -- 6（UTF-8中文占3字节）

-- CHAR_LENGTH：字符长度
SELECT CHAR_LENGTH('Hello');  -- 5
SELECT CHAR_LENGTH('你好');   -- 2
```

### 1.3 字符串截取

```sql
-- SUBSTRING：截取字符串
-- SUBSTRING(str, pos, len)
SELECT SUBSTRING('Hello World', 1, 5);   -- Hello
SELECT SUBSTRING('Hello World', 7);      -- World
SELECT SUBSTRING('Hello World', -5);     -- World

-- LEFT：从左边截取
SELECT LEFT('Hello World', 5);   -- Hello

-- RIGHT：从右边截取
SELECT RIGHT('Hello World', 5);  -- World
```

### 1.4 字符串查找

```sql
-- LOCATE：查找子串位置（返回位置，从1开始）
SELECT LOCATE('o', 'Hello World');     -- 5
SELECT LOCATE('World', 'Hello World'); -- 7
SELECT LOCATE('x', 'Hello World');     -- 0（未找到）

-- INSTR：查找子串位置（同LOCATE）
SELECT INSTR('Hello World', 'o');      -- 5

-- POSITION：查找子串位置
SELECT POSITION('World' IN 'Hello World');  -- 7
```

### 1.5 字符串替换

```sql
-- REPLACE：替换字符串
SELECT REPLACE('Hello World', 'World', 'MySQL');  -- Hello MySQL
SELECT REPLACE(name, '张', '王') FROM students;

-- INSERT：插入字符串
SELECT INSERT('Hello World', 7, 5, 'MySQL');  -- Hello MySQL
```

### 1.6 大小写转换

```sql
-- UPPER/UCASE：转大写
SELECT UPPER('hello');   -- HELLO
SELECT UCASE('hello');   -- HELLO

-- LOWER/LCASE：转小写
SELECT LOWER('HELLO');   -- hello
SELECT LCASE('HELLO');   -- hello
```

### 1.7 去除空格

```sql
-- TRIM：去除两端空格
SELECT TRIM('  Hello  ');        -- 'Hello'

-- LTRIM：去除左边空格
SELECT LTRIM('  Hello  ');       -- 'Hello  '

-- RTRIM：去除右边空格
SELECT RTRIM('  Hello  ');       -- '  Hello'

-- TRIM 指定字符
SELECT TRIM('x' FROM 'xxxHelloxxx');  -- 'Hello'
```

### 1.8 字符串填充

```sql
-- LPAD：左填充
SELECT LPAD('123', 5, '0');   -- 00123

-- RPAD：右填充
SELECT RPAD('123', 5, '0');   -- 12300
```

### 1.9 字符串重复

```sql
-- REPEAT：重复字符串
SELECT REPEAT('*', 5);   -- *****
SELECT REPEAT('AB', 3);  -- ABABAB
```

### 1.10 字符串反转

```sql
-- REVERSE：反转字符串
SELECT REVERSE('Hello');  -- olleH
```

## 二、数值函数

### 2.1 四舍五入

```sql
-- ROUND：四舍五入
SELECT ROUND(3.14159);        -- 3
SELECT ROUND(3.14159, 2);     -- 3.14
SELECT ROUND(3.14159, 4);     -- 3.1416

-- TRUNCATE：截断（不四舍五入）
SELECT TRUNCATE(3.14159, 2);  -- 3.14
```

### 2.2 向上/向下取整

```sql
-- CEIL/CEILING：向上取整
SELECT CEIL(3.14);      -- 4
SELECT CEILING(3.14);   -- 4
SELECT CEIL(-3.14);     -- -3

-- FLOOR：向下取整
SELECT FLOOR(3.14);     -- 3
SELECT FLOOR(-3.14);    -- -4
```

### 2.3 绝对值

```sql
-- ABS：绝对值
SELECT ABS(-10);   -- 10
SELECT ABS(10);    -- 10
```

### 2.4 符号函数

```sql
-- SIGN：返回符号（-1, 0, 1）
SELECT SIGN(-10);  -- -1
SELECT SIGN(0);    -- 0
SELECT SIGN(10);   -- 1
```

### 2.5 幂运算

```sql
-- POW/POWER：幂运算
SELECT POW(2, 3);    -- 8
SELECT POWER(2, 3);  -- 8

-- SQRT：平方根
SELECT SQRT(16);     -- 4
SELECT SQRT(2);      -- 1.4142135623730951
```

### 2.6 随机数

```sql
-- RAND：生成0-1之间的随机数
SELECT RAND();

-- 生成指定范围的随机数
SELECT FLOOR(RAND() * 100);           -- 0-99
SELECT FLOOR(RAND() * (100-50+1)) + 50;  -- 50-100
```

### 2.7 取模

```sql
-- MOD：取模（求余数）
SELECT MOD(10, 3);   -- 1
SELECT 10 % 3;       -- 1
```

## 三、日期时间函数

### 3.1 获取当前日期时间

```sql
-- NOW：当前日期和时间
SELECT NOW();  -- 2024-01-29 20:30:45

-- CURDATE/CURRENT_DATE：当前日期
SELECT CURDATE();        -- 2024-01-29
SELECT CURRENT_DATE();   -- 2024-01-29

-- CURTIME/CURRENT_TIME：当前时间
SELECT CURTIME();        -- 20:30:45
SELECT CURRENT_TIME();   -- 20:30:45

-- CURRENT_TIMESTAMP：当前时间戳
SELECT CURRENT_TIMESTAMP();  -- 2024-01-29 20:30:45
```

### 3.2 提取日期时间部分

```sql
-- YEAR：提取年份
SELECT YEAR('2024-01-29');   -- 2024
SELECT YEAR(NOW());

-- MONTH：提取月份
SELECT MONTH('2024-01-29');  -- 1

-- DAY：提取日
SELECT DAY('2024-01-29');    -- 29

-- HOUR：提取小时
SELECT HOUR('20:30:45');     -- 20

-- MINUTE：提取分钟
SELECT MINUTE('20:30:45');   -- 30

-- SECOND：提取秒
SELECT SECOND('20:30:45');   -- 45

-- EXTRACT：提取指定部分
SELECT EXTRACT(YEAR FROM '2024-01-29');   -- 2024
SELECT EXTRACT(MONTH FROM '2024-01-29');  -- 1
```

### 3.3 日期时间计算

```sql
-- DATE_ADD：日期加法
SELECT DATE_ADD('2024-01-29', INTERVAL 1 DAY);    -- 2024-01-30
SELECT DATE_ADD('2024-01-29', INTERVAL 1 MONTH);  -- 2024-02-29
SELECT DATE_ADD('2024-01-29', INTERVAL 1 YEAR);   -- 2025-01-29
SELECT DATE_ADD('2024-01-29', INTERVAL -1 DAY);   -- 2024-01-28

-- DATE_SUB：日期减法
SELECT DATE_SUB('2024-01-29', INTERVAL 1 DAY);    -- 2024-01-28
SELECT DATE_SUB('2024-01-29', INTERVAL 1 MONTH);  -- 2023-12-29

-- ADDDATE：日期加法（同DATE_ADD）
SELECT ADDDATE('2024-01-29', 7);  -- 2024-02-05（加7天）

-- SUBDATE：日期减法（同DATE_SUB）
SELECT SUBDATE('2024-01-29', 7);  -- 2024-01-22（减7天）
```

### 3.4 日期差值

```sql
-- DATEDIFF：计算日期差（天数）
SELECT DATEDIFF('2024-01-29', '2024-01-01');  -- 28
SELECT DATEDIFF('2024-01-01', '2024-01-29');  -- -28

-- TIMEDIFF：计算时间差
SELECT TIMEDIFF('20:30:45', '18:00:00');  -- 02:30:45
```

### 3.5 日期格式化

```sql
-- DATE_FORMAT：格式化日期
SELECT DATE_FORMAT('2024-01-29 20:30:45', '%Y-%m-%d');  -- 2024-01-29
SELECT DATE_FORMAT('2024-01-29 20:30:45', '%Y年%m月%d日');  -- 2024年01月29日
SELECT DATE_FORMAT('2024-01-29 20:30:45', '%H:%i:%s');  -- 20:30:45
SELECT DATE_FORMAT('2024-01-29 20:30:45', '%W');  -- Monday

-- 常用格式符号
-- %Y：4位年份    %y：2位年份
-- %m：月份(01-12)  %c：月份(1-12)
-- %d：日期(01-31)  %e：日期(1-31)
-- %H：小时(00-23)  %h：小时(01-12)
-- %i：分钟(00-59)
-- %s：秒(00-59)
-- %W：星期名称    %w：星期(0-6)
```

### 3.6 字符串转日期

```sql
-- STR_TO_DATE：字符串转日期
SELECT STR_TO_DATE('2024-01-29', '%Y-%m-%d');
SELECT STR_TO_DATE('29/01/2024', '%d/%m/%Y');
SELECT STR_TO_DATE('2024年01月29日', '%Y年%m月%d日');
```

### 3.7 时间戳

```sql
-- UNIX_TIMESTAMP：转换为时间戳
SELECT UNIX_TIMESTAMP();  -- 当前时间戳
SELECT UNIX_TIMESTAMP('2024-01-29 20:30:45');

-- FROM_UNIXTIME：时间戳转日期
SELECT FROM_UNIXTIME(1706533845);
SELECT FROM_UNIXTIME(1706533845, '%Y-%m-%d %H:%i:%s');
```

### 3.8 星期和季度

```sql
-- DAYOFWEEK：星期几（1=周日，7=周六）
SELECT DAYOFWEEK('2024-01-29');  -- 2（周一）

-- WEEKDAY：星期几（0=周一，6=周日）
SELECT WEEKDAY('2024-01-29');    -- 0（周一）

-- DAYNAME：星期名称
SELECT DAYNAME('2024-01-29');    -- Monday

-- QUARTER：季度
SELECT QUARTER('2024-01-29');    -- 1
```

## 四、条件函数

### 4.1 IF 函数

```sql
-- IF(condition, true_value, false_value)
SELECT IF(10 > 5, 'Yes', 'No');  -- Yes

SELECT 
    name,
    age,
    IF(age >= 18, '成年', '未成年') AS age_status
FROM students;
```

### 4.2 IFNULL 函数

```sql
-- IFNULL(value, default_value)
SELECT IFNULL(NULL, 'Default');  -- Default
SELECT IFNULL('Value', 'Default');  -- Value

SELECT 
    name,
    IFNULL(email, '未填写') AS email
FROM students;
```

### 4.3 NULLIF 函数

```sql
-- NULLIF(value1, value2)
-- 如果相等返回NULL，否则返回value1
SELECT NULLIF(10, 10);  -- NULL
SELECT NULLIF(10, 20);  -- 10
```

### 4.4 CASE WHEN 表达式

```sql
-- 简单CASE表达式
SELECT 
    name,
    age,
    CASE age
        WHEN 18 THEN '刚成年'
        WHEN 19 THEN '19岁'
        WHEN 20 THEN '20岁'
        ELSE '其他'
    END AS age_desc
FROM students;

-- 搜索CASE表达式（推荐）
SELECT 
    name,
    age,
    CASE 
        WHEN age < 18 THEN '未成年'
        WHEN age >= 18 AND age < 25 THEN '青年'
        WHEN age >= 25 AND age < 40 THEN '中年'
        ELSE '老年'
    END AS age_group
FROM students;

-- 多条件CASE
SELECT 
    name,
    age,
    gender,
    CASE 
        WHEN age >= 18 AND gender = '男' THEN '成年男性'
        WHEN age >= 18 AND gender = '女' THEN '成年女性'
        WHEN age < 18 AND gender = '男' THEN '未成年男性'
        WHEN age < 18 AND gender = '女' THEN '未成年女性'
        ELSE '未知'
    END AS category
FROM students;
```

## 六、运算符

### 6.1 算术运算符

```sql
-- 加法
SELECT 10 + 5;   -- 15

-- 减法
SELECT 10 - 5;   -- 5

-- 乘法
SELECT 10 * 5;   -- 50

-- 除法
SELECT 10 / 5;   -- 2.0000
SELECT 10 / 3;   -- 3.3333

-- 整除
SELECT 10 DIV 3; -- 3

-- 取模
SELECT 10 % 3;   -- 1
SELECT MOD(10, 3);  -- 1
```

### 6.2 比较运算符

```sql
-- 等于
SELECT 10 = 10;   -- 1 (true)
SELECT 10 = 5;    -- 0 (false)

-- 不等于
SELECT 10 != 5;   -- 1
SELECT 10 <> 5;   -- 1

-- 大于
SELECT 10 > 5;    -- 1

-- 小于
SELECT 10 < 5;    -- 0

-- 大于等于
SELECT 10 >= 10;  -- 1

-- 小于等于
SELECT 10 <= 5;   -- 0

-- 安全等于（可以比较NULL）
SELECT NULL <=> NULL;  -- 1
SELECT 10 <=> NULL;    -- 0
```

### 6.3 逻辑运算符

```sql
-- AND（与）
SELECT 1 AND 1;   -- 1
SELECT 1 AND 0;   -- 0

-- OR（或）
SELECT 1 OR 0;    -- 1
SELECT 0 OR 0;    -- 0

-- NOT（非）
SELECT NOT 1;     -- 0
SELECT NOT 0;     -- 1

-- XOR（异或）
SELECT 1 XOR 1;   -- 0
SELECT 1 XOR 0;   -- 1
```

### 6.4 位运算符

```sql
-- 位与
SELECT 5 & 3;     -- 1

-- 位或
SELECT 5 | 3;     -- 7

-- 位异或
SELECT 5 ^ 3;     -- 6

-- 位取反
SELECT ~5;        -- -6

-- 左移
SELECT 5 << 1;    -- 10

-- 右移
SELECT 5 >> 1;    -- 2
```

## 七、实战练习

### 练习1：字符串函数

```sql
-- 1. 拼接学生姓名和年龄
SELECT CONCAT(name, '(', age, '岁)') AS info FROM students;

-- 2. 查询姓名长度大于2的学生
SELECT * FROM students WHERE CHAR_LENGTH(name) > 2;

-- 3. 将学生姓名转为大写
SELECT UPPER(name) FROM students;

-- 4. 截取邮箱的用户名部分
SELECT 
    email,
    SUBSTRING(email, 1, LOCATE('@', email) - 1) AS username
FROM students
WHERE email IS NOT NULL;
```

### 练习2：数值函数

```sql
-- 1. 四舍五入成绩
SELECT 
    student_id,
    course_id,
    score,
    ROUND(score) AS rounded_score
FROM scores;

-- 2. 生成1-100的随机数
SELECT FLOOR(RAND() * 100) + 1 AS random_num;

-- 3. 计算成绩的绝对差值
SELECT 
    score,
    ABS(score - 85) AS diff_from_85
FROM scores;
```

### 练习3：日期函数

```sql
-- 1. 查询今天创建的学生
SELECT * FROM students
WHERE DATE(create_time) = CURDATE();

-- 2. 查询最近7天创建的学生
SELECT * FROM students
WHERE create_time >= DATE_SUB(NOW(), INTERVAL 7 DAY);

-- 3. 计算学生创建天数
SELECT 
    name,
    create_time,
    DATEDIFF(NOW(), create_time) AS days_since_created
FROM students;

-- 4. 格式化创建时间
SELECT 
    name,
    DATE_FORMAT(create_time, '%Y年%m月%d日 %H:%i') AS formatted_time
FROM students;
```

### 练习4：条件函数

```sql
-- 1. 根据年龄判断是否成年
SELECT 
    name,
    age,
    IF(age >= 18, '成年', '未成年') AS status
FROM students;

-- 2. 根据成绩评级
SELECT 
    student_id,
    course_id,
    score,
    CASE 
        WHEN score >= 90 THEN '优秀'
        WHEN score >= 80 THEN '良好'
        WHEN score >= 70 THEN '中等'
        WHEN score >= 60 THEN '及格'
        ELSE '不及格'
    END AS grade
FROM scores;

-- 3. 处理NULL值
SELECT 
    name,
    IFNULL(email, '未填写邮箱') AS email
FROM students;
```
