# 存储过程与函数

## 一、存储过程概述

### 1.1 什么是存储过程

存储过程是一组预编译的 SQL 语句集合，存储在数据库中，可以重复调用。

**优点**：
- 提高性能（预编译）
- 减少网络传输
- 提高安全性
- 代码复用
- 业务逻辑封装

**缺点**：
- 调试困难
- 移植性差
- 维护成本高
- 版本管理困难

### 1.2 存储过程 vs SQL 语句

| 特性 | 存储过程 | SQL 语句 |
|------|----------|----------|
| 执行方式 | 预编译 | 每次编译 |
| 网络传输 | 少 | 多 |
| 安全性 | 高 | 一般 |
| 复用性 | 高 | 低 |
| 调试 | 困难 | 容易 |

## 二、创建存储过程

### 2.1 基本语法

```sql
DELIMITER $$

CREATE PROCEDURE procedure_name(
    [IN | OUT | INOUT] parameter_name datatype,
    ...
)
BEGIN
    -- SQL 语句
END$$

DELIMITER ;
```

### 2.2 简单存储过程

```sql
-- 修改分隔符
DELIMITER $$

-- 创建存储过程
CREATE PROCEDURE get_student_count()
BEGIN
    SELECT COUNT(*) AS total FROM students;
END$$

-- 恢复分隔符
DELIMITER ;

-- 调用存储过程
CALL get_student_count();
```

### 2.3 带参数的存储过程

```sql
-- IN 参数（输入参数）
DELIMITER $$

CREATE PROCEDURE get_students_by_age(IN age_param INT)
BEGIN
    SELECT * FROM students WHERE age = age_param;
END$$

DELIMITER ;

-- 调用
CALL get_students_by_age(20);

-- OUT 参数（输出参数）
DELIMITER $$

CREATE PROCEDURE get_student_count_by_gender(
    IN gender_param VARCHAR(10),
    OUT count_result INT
)
BEGIN
    SELECT COUNT(*) INTO count_result 
    FROM students 
    WHERE gender = gender_param;
END$$

DELIMITER ;

-- 调用
CALL get_student_count_by_gender('男', @male_count);
SELECT @male_count;

-- INOUT 参数（输入输出参数）
DELIMITER $$

CREATE PROCEDURE increment_age(INOUT age_param INT)
BEGIN
    SET age_param = age_param + 1;
END$$

DELIMITER ;

-- 调用
SET @age = 20;
CALL increment_age(@age);
SELECT @age;  -- 21
```

## 三、变量

### 3.1 局部变量

```sql
DELIMITER $$

CREATE PROCEDURE variable_demo()
BEGIN
    -- 声明变量
    DECLARE var_name VARCHAR(50);
    DECLARE var_age INT DEFAULT 0;
    DECLARE var_count INT;
    
    -- 赋值
    SET var_name = '张三';

CALL variable_demo();
```

## 四、流程控制

### 4.1 IF 语句

```sql
DELIMITER $$

CREATE PROCEDURE check_age(IN age_param INT)
BEGIN
    IF age_param >= 18 THEN
        SELECT '成年人' AS result;
    ELSE
        SELECT '未成年人' AS result;
    END IF;
END$$

DELIMITER ;

CALL check_age(20);

-- IF...ELSEIF...ELSE
DELIMITER $$

CREATE PROCEDURE get_grade(IN score_param DECIMAL(5,2))
BEGIN
    DECLARE grade VARCHAR(10);
    
    IF score_param >= 90 THEN
        SET grade = '优秀';
    ELSEIF score_param >= 80 THEN
        SET grade = '良好';
    ELSEIF score_param >= 70 THEN
        SET grade = '中等';
    ELSEIF score_param >= 60 THEN
        SET grade = '及格';
    ELSE
        SET grade = '不及格';
    END IF;
    
    SELECT grade;
END$$

DELIMITER ;

CALL get_grade(85);
```

### 4.2 CASE 语句

```sql
DELIMITER $$

CREATE PROCEDURE get_weekday(IN day_num INT)
BEGIN
    DECLARE day_name VARCHAR(20);
    
    CASE day_num
        WHEN 1 THEN SET day_name = '星期一';
        WHEN 2 THEN SET day_name = '星期二';
        WHEN 3 THEN SET day_name = '星期三';
        WHEN 4 THEN SET day_name = '星期四';
        WHEN 5 THEN SET day_name = '星期五';
        WHEN 6 THEN SET day_name = '星期六';
        WHEN 7 THEN SET day_name = '星期日';
        ELSE SET day_name = '无效';
    END CASE;
    
    SELECT day_name;
END$$

DELIMITER ;

CALL get_weekday(3);
```

### 4.3 WHILE 循环

```sql
DELIMITER $$

CREATE PROCEDURE sum_numbers(IN n INT)
BEGIN
    DECLARE i INT DEFAULT 1;
    DECLARE total INT DEFAULT 0;
    
    WHILE i <= n DO
        SET total = total + i;
        SET i = i + 1;
    END WHILE;
    
    SELECT total;
END$$

DELIMITER ;

CALL sum_numbers(100);  -- 5050
```

### 4.4 REPEAT 循环

```sql
DELIMITER $$

CREATE PROCEDURE repeat_demo(IN n INT)
BEGIN
    DECLARE i INT DEFAULT 1;
    DECLARE total INT DEFAULT 0;
    
    REPEAT
        SET total = total + i;
        SET i = i + 1;
    UNTIL i > n
    END REPEAT;
    
    SELECT total;
END$$

DELIMITER ;

CALL repeat_demo(100);
```

### 4.5 LOOP 循环

```sql
DELIMITER $$

CREATE PROCEDURE loop_demo(IN n INT)
BEGIN
    DECLARE i INT DEFAULT 1;
    DECLARE total INT DEFAULT 0;
    
    loop_label: LOOP
        SET total = total + i;
        SET i = i + 1;
        
        IF i > n THEN
            LEAVE loop_label;  -- 退出循环
        END IF;
    END LOOP;
    
    SELECT total;
END$$

DELIMITER ;

CALL loop_demo(100);
```

### 4.6 ITERATE（继续循环）

```sql
DELIMITER $$

CREATE PROCEDURE iterate_demo(IN n INT)
BEGIN
    DECLARE i INT DEFAULT 0;
    DECLARE total INT DEFAULT 0;
    
    loop_label: LOOP
        SET i = i + 1;
        
        -- 跳过偶数
        IF i % 2 = 0 THEN
            ITERATE loop_label;
        END IF;
        
        SET total = total + i;
        
        IF i >= n THEN
            LEAVE loop_label;
        END IF;
    END LOOP;
    
    SELECT total;  -- 奇数之和
END$$

DELIMITER ;

CALL iterate_demo(10);  -- 1+3+5+7+9 = 25
```

## 五、游标（CURSOR）

### 5.1 游标基础

```sql
DELIMITER $$

CREATE PROCEDURE cursor_demo()
BEGIN
    -- 声明变量
    DECLARE done INT DEFAULT FALSE;
    DECLARE v_id INT;
    DECLARE v_name VARCHAR(50);
    DECLARE v_age INT;
    
    -- 声明游标
    DECLARE student_cursor CURSOR FOR
        SELECT id, name, age FROM students;
    
    -- 声明 NOT FOUND 处理程序
    DECLARE CONTINUE HANDLER FOR NOT FOUND SET done = TRUE;
    
    -- 打开游标
    OPEN student_cursor;
    
    -- 循环读取数据
    read_loop: LOOP
        FETCH student_cursor INTO v_id, v_name, v_age;
        
        IF done THEN
            LEAVE read_loop;
        END IF;
        
        -- 处理数据
        SELECT v_id, v_name, v_age;
    END LOOP;
    
    -- 关闭游标
    CLOSE student_cursor;
END$$

DELIMITER ;

CALL cursor_demo();
```

### 5.2 游标实战案例

```sql
-- 案例：批量更新学生等级
DELIMITER $$

CREATE PROCEDURE update_student_level()
BEGIN
    DECLARE done INT DEFAULT FALSE;
    DECLARE v_id INT;
    DECLARE v_avg_score DECIMAL(5,2);
    DECLARE v_level VARCHAR(20);
    
    DECLARE score_cursor CURSOR FOR
        SELECT student_id, AVG(score) AS avg_score
        FROM scores
        GROUP BY student_id;
    
    DECLARE CONTINUE HANDLER FOR NOT FOUND SET done = TRUE;
    
    OPEN score_cursor;
    
    read_loop: LOOP
        FETCH score_cursor INTO v_id, v_avg_score;
        
        IF done THEN
            LEAVE read_loop;
        END IF;
        
        -- 根据平均分设置等级
        IF v_avg_score >= 90 THEN
            SET v_level = '优秀';
        ELSEIF v_avg_score >= 80 THEN
            SET v_level = '良好';
        ELSEIF v_avg_score >= 70 THEN
            SET v_level = '中等';
        ELSEIF v_avg_score >= 60 THEN
            SET v_level = '及格';
        ELSE
            SET v_level = '不及格';
        END IF;
        
        -- 更新学生等级
        UPDATE students SET level = v_level WHERE id = v_id;
    END LOOP;
    
    CLOSE score_cursor;
    
    SELECT '更新完成' AS result;
END$$

DELIMITER ;

CALL update_student_level();
```

### 6.1 HANDLER 声明

```sql
DELIMITER $$

CREATE PROCEDURE error_handler_demo()
BEGIN
    -- 声明异常处理程序
    DECLARE CONTINUE HANDLER FOR SQLEXCEPTION

        SELECT '发生错误' AS message;

END$$

```

### 6.2 异常处理类型

```sql
DELIMITER $$

CREATE PROCEDURE handler_types_demo()
BEGIN
    -- CONTINUE：继续执行
    DECLARE CONTINUE HANDLER FOR SQLEXCEPTION
        SELECT '错误：继续执行' AS message;
    
    -- EXIT：退出当前代码块
    -- DECLARE EXIT HANDLER FOR SQLEXCEPTION
    --     SELECT '错误：退出执行' AS message;
    
    -- UNDO：回滚（MySQL 不支持）
    
    INSERT INTO students (id, name) VALUES (1, '张三');
    SELECT '第一条语句执行完成';
```

## 七、自定义函数

### 7.1 创建函数

```sql
DELIMITER $$

CREATE FUNCTION get_student_name(student_id INT)
RETURNS VARCHAR(50)
DETERMINISTIC
BEGIN
    DECLARE student_name VARCHAR(50);
    
    SELECT name INTO student_name
    FROM students
    WHERE id = student_id;
    
    RETURN student_name;
END$$

DELIMITER ;

-- 使用函数
SELECT get_student_name(1);
SELECT id, get_student_name(id) AS name FROM scores;
```

### 7.2 函数特性

```sql
-- DETERMINISTIC：相同输入总是返回相同输出
-- NOT DETERMINISTIC：可能返回不同输出（默认）
-- READS SQL DATA：读取数据
-- MODIFIES SQL DATA：修改数据
-- NO SQL：不包含 SQL 语句

DELIMITER $$

CREATE FUNCTION calculate_age(birth_date DATE)
RETURNS INT
DETERMINISTIC
READS SQL DATA
BEGIN
    RETURN YEAR(CURDATE()) - YEAR(birth_date);
END$$

DELIMITER ;

SELECT calculate_age('2000-01-01');
```

### 7.3 函数实战案例

```sql
-- 案例1：计算学生平均分
DELIMITER $$

CREATE FUNCTION get_avg_score(student_id INT)
RETURNS DECIMAL(5,2)
READS SQL DATA
BEGIN
    DECLARE avg_score DECIMAL(5,2);
    
    SELECT AVG(score) INTO avg_score
    FROM scores
    WHERE student_id = student_id;
    
    RETURN IFNULL(avg_score, 0);
END$$

DELIMITER ;

SELECT id, name, get_avg_score(id) AS avg_score FROM students;

-- 案例2：判断成绩等级
DELIMITER $$

CREATE FUNCTION get_score_grade(score DECIMAL(5,2))
RETURNS VARCHAR(10)
DETERMINISTIC
BEGIN
    IF score >= 90 THEN
        RETURN '优秀';
    ELSEIF score >= 80 THEN
        RETURN '良好';
    ELSEIF score >= 70 THEN
        RETURN '中等';
    ELSEIF score >= 60 THEN
        RETURN '及格';
    ELSE
        RETURN '不及格';
    END IF;
END$$

DELIMITER ;

SELECT student_id, course_id, score, get_score_grade(score) AS grade
FROM scores;
```


## 九、实战案例

### 案例1：订单处理存储过程

```sql
DELIMITER $$

CREATE PROCEDURE process_order(
    IN p_user_id INT,
    IN p_product_id INT,
    IN p_quantity INT,
    OUT p_order_id INT,
    OUT p_message VARCHAR(200)
)
BEGIN
    DECLARE v_stock INT;
    DECLARE v_price DECIMAL(10,2);
    DECLARE v_total DECIMAL(10,2);
    
    -- 声明异常处理
    DECLARE EXIT HANDLER FOR SQLEXCEPTION
    BEGIN
        ROLLBACK;
        SET p_message = '订单处理失败';
    END;
    
    -- 开始事务
    START TRANSACTION;
    
    -- 检查库存
    SELECT stock, price INTO v_stock, v_price
    FROM products
    WHERE id = p_product_id
    FOR UPDATE;
    
    IF v_stock < p_quantity THEN
        SET p_message = '库存不足';
        ROLLBACK;
    ELSE
        -- 计算总价
        SET v_total = v_price * p_quantity;
        
        -- 创建订单
        INSERT INTO orders (user_id, product_id, quantity, total_price, status)
        VALUES (p_user_id, p_product_id, p_quantity, v_total, 'pending');
        
        SET p_order_id = LAST_INSERT_ID();
        
        -- 扣减库存
        UPDATE products
        SET stock = stock - p_quantity
        WHERE id = p_product_id;
        
        -- 提交事务
        COMMIT;
        SET p_message = '订单创建成功';
    END IF;
END$$

DELIMITER ;

-- 调用
CALL process_order(1, 100, 2, @order_id, @message);
SELECT @order_id, @message;
```

### 案例2：数据统计存储过程

```sql
DELIMITER $$

CREATE PROCEDURE get_class_statistics(IN p_class_id INT)
BEGIN
    -- 学生总数
    SELECT COUNT(*) AS student_count
    FROM students
    WHERE class_id = p_class_id;
    
    -- 平均年龄
    SELECT AVG(age) AS avg_age
    FROM students
    WHERE class_id = p_class_id;
    
    -- 性别分布
    SELECT gender, COUNT(*) AS count
    FROM students
    WHERE class_id = p_class_id
    GROUP BY gender;
    
    -- 平均成绩
    SELECT AVG(sc.score) AS avg_score
    FROM students s
    JOIN scores sc ON s.id = sc.student_id
    WHERE s.class_id = p_class_id;
END$$

DELIMITER ;

CALL get_class_statistics(1);
```
