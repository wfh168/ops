## 1. 触发器概述

### 1.1 什么是触发器

触发器（Trigger）是与表相关联的数据库对象，当对表进行 INSERT、UPDATE、DELETE 操作时自动执行。

**特点**：
- 自动执行，无需手动调用
- 与表紧密关联
- 可以在操作前后执行
- 可以访问操作的新旧数据

### 1.2 触发器的作用

```
✅ 数据完整性约束
✅ 业务规则实施
✅ 数据审计和日志
✅ 数据同步
✅ 自动计算和更新
```

### 1.3 触发器的类型

**按触发时机分类**：
- BEFORE 触发器：在操作执行前触发
- AFTER 触发器：在操作执行后触发

**按触发事件分类**：
- INSERT 触发器
- UPDATE 触发器
- DELETE 触发器

## 2. 创建触发器

### 2.1 基本语法

```sql
CREATE TRIGGER trigger_name
trigger_time trigger_event
ON table_name
FOR EACH ROW
trigger_body

-- trigger_time: BEFORE 或 AFTER
-- trigger_event: INSERT、UPDATE 或 DELETE
-- FOR EACH ROW: 行级触发器
```

### 2.2 INSERT 触发器

**示例1：记录新增用户**

```sql
-- 创建用户表
CREATE TABLE users (
    id INT PRIMARY KEY AUTO_INCREMENT,
    username VARCHAR(50) NOT NULL,
    email VARCHAR(100),
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- 创建用户日志表
CREATE TABLE user_logs (
    id INT PRIMARY KEY AUTO_INCREMENT,
    user_id INT,
    action VARCHAR(50),
    log_time DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- 创建 INSERT 触发器
DELIMITER $$
CREATE TRIGGER after_user_insert
AFTER INSERT ON users
FOR EACH ROW
BEGIN
    INSERT INTO user_logs (user_id, action)
    VALUES (NEW.id, 'User Created');
END$$
DELIMITER ;

-- 测试触发器
INSERT INTO users (username, email) 
VALUES ('zhangsan', 'zhangsan@example.com');

-- 查看日志
SELECT * FROM user_logs;
```

**示例2：自动生成订单号**

```sql
-- 创建订单表
CREATE TABLE orders (
    id INT PRIMARY KEY AUTO_INCREMENT,
    order_no VARCHAR(50),
    user_id INT,
    amount DECIMAL(10,2),
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- 创建 BEFORE INSERT 触发器
DELIMITER $$
CREATE TRIGGER before_order_insert
BEFORE INSERT ON orders
FOR EACH ROW
BEGIN
    -- 生成订单号：ORD + 日期 + 随机数
    SET NEW.order_no = CONCAT(
        'ORD',
        DATE_FORMAT(NOW(), '%Y%m%d'),
        LPAD(FLOOR(RAND() * 10000), 4, '0')
    );
END$$
DELIMITER ;

-- 测试
INSERT INTO orders (user_id, amount) VALUES (1, 99.99);
SELECT * FROM orders;
```

### 2.3 UPDATE 触发器

**示例1：记录数据变更**

```sql
-- 创建商品表
CREATE TABLE products (
    id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(100),
    price DECIMAL(10,2),
    stock INT,
    updated_at DATETIME
);

-- 创建变更历史表
CREATE TABLE product_history (
    id INT PRIMARY KEY AUTO_INCREMENT,
    product_id INT,
    old_price DECIMAL(10,2),
    new_price DECIMAL(10,2),
    old_stock INT,
    new_stock INT,
    change_time DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- 创建 UPDATE 触发器
DELIMITER $$
CREATE TRIGGER after_product_update
AFTER UPDATE ON products
FOR EACH ROW
BEGIN
    -- 只记录价格或库存变化
    IF OLD.price != NEW.price OR OLD.stock != NEW.stock THEN
        INSERT INTO product_history (
            product_id, old_price, new_price, 
            old_stock, new_stock
        ) VALUES (
            NEW.id, OLD.price, NEW.price,
            OLD.stock, NEW.stock
        );
    END IF;
END$$
DELIMITER ;

-- 测试
INSERT INTO products (name, price, stock) 
VALUES ('iPhone 15', 5999.00, 100);

UPDATE products SET price = 5499.00 WHERE id = 1;
UPDATE products SET stock = 95 WHERE id = 1;

SELECT * FROM product_history;
```

**示例2：自动更新时间戳**

```sql
DELIMITER $$
CREATE TRIGGER before_product_update
BEFORE UPDATE ON products
FOR EACH ROW
BEGIN
    SET NEW.updated_at = NOW();
END$$
DELIMITER ;
```

### 2.4 DELETE 触发器

**示例1：软删除实现**

```sql
-- 创建员工表
CREATE TABLE employees (
    id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(50),
    department VARCHAR(50),
    salary DECIMAL(10,2),
    is_deleted TINYINT DEFAULT 0,
    deleted_at DATETIME
);

-- 创建已删除员工表
CREATE TABLE deleted_employees (
    id INT PRIMARY KEY AUTO_INCREMENT,
    employee_id INT,
    name VARCHAR(50),
    department VARCHAR(50),
    salary DECIMAL(10,2),
    deleted_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- 创建 DELETE 触发器
DELIMITER $$
CREATE TRIGGER before_employee_delete
BEFORE DELETE ON employees
FOR EACH ROW
BEGIN
    -- 备份到已删除表
    INSERT INTO deleted_employees (
        employee_id, name, department, salary
    ) VALUES (
        OLD.id, OLD.name, OLD.department, OLD.salary
    );
END$$
DELIMITER ;

-- 测试
INSERT INTO employees (name, department, salary)
VALUES ('李四', '技术部', 8000);

DELETE FROM employees WHERE id = 1;

SELECT * FROM deleted_employees;
```

**示例2：级联删除相关数据**

```sql
-- 创建订单明细表
CREATE TABLE order_items (
    id INT PRIMARY KEY AUTO_INCREMENT,
    order_id INT,
    product_id INT,
    quantity INT,
    price DECIMAL(10,2)
);

-- 创建触发器：删除订单时自动删除订单明细
DELIMITER $$
CREATE TRIGGER after_order_delete
AFTER DELETE ON orders
FOR EACH ROW
BEGIN
    DELETE FROM order_items WHERE order_id = OLD.id;
END$$
DELIMITER ;
```

## 3. NEW 和 OLD 关键字

### 3.1 使用规则

| 触发器类型 | NEW | OLD |
|-----------|-----|-----|
| INSERT | ✅ 可用 | ❌ 不可用 |
| UPDATE | ✅ 可用 | ✅ 可用 |
| DELETE | ❌ 不可用 | ✅ 可用 |

### 3.2 实际应用

```sql
DELIMITER $$
CREATE TRIGGER example_trigger
BEFORE UPDATE ON products
FOR EACH ROW
BEGIN
    -- OLD：更新前的值
    -- NEW：更新后的值
    
    -- 防止价格降低超过50%
    IF NEW.price < OLD.price * 0.5 THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = '价格降幅不能超过50%';
    END IF;
    
    -- 记录价格变化百分比
    SET @price_change = (NEW.price - OLD.price) / OLD.price * 100;
END$$
DELIMITER ;
```

## 4. 触发器管理

### 4.1 查看触发器

```sql
-- 查看所有触发器
SHOW TRIGGERS;

-- 查看指定数据库的触发器
SHOW TRIGGERS FROM database_name;

-- 查看触发器详细信息
SHOW CREATE TRIGGER trigger_name;

-- 从系统表查询
SELECT * FROM information_schema.TRIGGERS
WHERE TRIGGER_SCHEMA = 'your_database';
```

### 4.2 删除触发器

```sql
-- 删除触发器
DROP TRIGGER IF EXISTS trigger_name;

-- 示例
DROP TRIGGER IF EXISTS after_user_insert;
```

### 4.3 修改触发器

```sql
-- MySQL 不支持直接修改触发器
-- 需要先删除再重建

DROP TRIGGER IF EXISTS after_user_insert;

DELIMITER $$
CREATE TRIGGER after_user_insert
AFTER INSERT ON users
FOR EACH ROW
BEGIN
    -- 新的触发器逻辑
    INSERT INTO user_logs (user_id, action, details)
    VALUES (NEW.id, 'User Created', CONCAT('Email: ', NEW.email));
END$$
DELIMITER ;
```

## 5. 触发器应用场景

### 5.1 数据审计

```sql
-- 创建审计表
CREATE TABLE audit_log (
    id INT PRIMARY KEY AUTO_INCREMENT,
    table_name VARCHAR(50),
    operation VARCHAR(20),
    user VARCHAR(50),
    old_data TEXT,
    new_data TEXT,
    change_time DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- 创建审计触发器
DELIMITER $$
CREATE TRIGGER audit_products_update
AFTER UPDATE ON products
FOR EACH ROW
BEGIN
    INSERT INTO audit_log (
        table_name, operation, user, old_data, new_data
    ) VALUES (
        'products',
        'UPDATE',
        USER(),
        CONCAT('price:', OLD.price, ',stock:', OLD.stock),
        CONCAT('price:', NEW.price, ',stock:', NEW.stock)
    );
END$$
DELIMITER ;
```

### 5.2 数据验证

```sql
-- 库存不能为负数
DELIMITER $$
CREATE TRIGGER check_stock_before_update
BEFORE UPDATE ON products
FOR EACH ROW
BEGIN
    IF NEW.stock < 0 THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = '库存不能为负数';
    END IF;
END$$
DELIMITER ;
```

### 5.3 自动计算

```sql
-- 创建订单表
CREATE TABLE orders_with_total (
    id INT PRIMARY KEY AUTO_INCREMENT,
    user_id INT,
    subtotal DECIMAL(10,2),
    tax DECIMAL(10,2),
    total DECIMAL(10,2)
);

-- 自动计算总价
DELIMITER $$
CREATE TRIGGER calculate_order_total
BEFORE INSERT ON orders_with_total
FOR EACH ROW
BEGIN
    SET NEW.tax = NEW.subtotal * 0.13;  -- 13% 税率
    SET NEW.total = NEW.subtotal + NEW.tax;
END$$
DELIMITER ;

-- 测试
INSERT INTO orders_with_total (user_id, subtotal)
VALUES (1, 100.00);

SELECT * FROM orders_with_total;
```

### 5.4 数据同步

```sql
-- 主表
CREATE TABLE main_products (
    id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(100),
    price DECIMAL(10,2)
);

-- 缓存表
CREATE TABLE cache_products (
    id INT PRIMARY KEY,
    name VARCHAR(100),
    price DECIMAL(10,2),
    sync_time DATETIME
);

-- 同步触发器
DELIMITER $$
CREATE TRIGGER sync_products_after_insert
AFTER INSERT ON main_products
FOR EACH ROW
BEGIN
    INSERT INTO cache_products (id, name, price, sync_time)
    VALUES (NEW.id, NEW.name, NEW.price, NOW())
    ON DUPLICATE KEY UPDATE
        name = NEW.name,
        price = NEW.price,
        sync_time = NOW();
END$$
DELIMITER ;
```

## 6. 触发器注意事项

### 6.1 性能影响

```
⚠️ 触发器会增加数据库操作的开销
⚠️ 复杂的触发器会降低性能
⚠️ 大量数据操作时影响更明显
```

**优化建议**：
```sql
-- ❌ 不好的做法：触发器中执行复杂查询
DELIMITER $$
CREATE TRIGGER bad_trigger
AFTER INSERT ON orders
FOR EACH ROW
BEGIN
    -- 复杂的统计查询
    SELECT COUNT(*), SUM(amount) INTO @cnt, @total
    FROM orders WHERE user_id = NEW.user_id;
    
    -- 更新用户统计
    UPDATE user_stats SET order_count = @cnt, total_amount = @total
    WHERE user_id = NEW.user_id;
END$$
DELIMITER ;

-- ✅ 好的做法：简单快速的操作
DELIMITER $$
CREATE TRIGGER good_trigger
AFTER INSERT ON orders
FOR EACH ROW
BEGIN
    -- 简单的增量更新
    UPDATE user_stats 
    SET order_count = order_count + 1,
        total_amount = total_amount + NEW.amount
    WHERE user_id = NEW.user_id;
END$$
DELIMITER ;
```

### 6.2 触发器限制

```
❌ 不能返回结果集
❌ 不能使用 CALL 调用返回结果集的存储过程
❌ 不能使用显式或隐式的事务控制语句
❌ 同一个表的同一事件只能有一个触发器
```

### 6.3 调试困难

```sql
-- 使用日志表辅助调试
CREATE TABLE trigger_debug_log (
    id INT PRIMARY KEY AUTO_INCREMENT,
    trigger_name VARCHAR(50),
    message TEXT,
    log_time DATETIME DEFAULT CURRENT_TIMESTAMP
);

DELIMITER $$
CREATE TRIGGER debug_example
BEFORE UPDATE ON products
FOR EACH ROW
BEGIN
    -- 记录调试信息
    INSERT INTO trigger_debug_log (trigger_name, message)
    VALUES ('debug_example', 
            CONCAT('Old price:', OLD.price, ', New price:', NEW.price));
END$$
DELIMITER ;
```

### 6.4 触发器链

```sql
-- 避免触发器链式反应
-- 表A的触发器修改表B
-- 表B的触发器修改表C
-- 可能导致性能问题或死循环

-- ❌ 不好的设计
CREATE TRIGGER trigger_a AFTER UPDATE ON table_a
FOR EACH ROW
BEGIN
    UPDATE table_b SET col = NEW.val WHERE id = NEW.id;
END;

CREATE TRIGGER trigger_b AFTER UPDATE ON table_b
FOR EACH ROW
BEGIN
    UPDATE table_c SET col = NEW.val WHERE id = NEW.id;
END;
```

## 7. 实战案例

### 案例1：电商库存管理

```sql
-- 商品表
CREATE TABLE shop_products (
    id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(100),
    stock INT,
    sold INT DEFAULT 0
);

-- 订单表
CREATE TABLE shop_orders (
    id INT PRIMARY KEY AUTO_INCREMENT,
    product_id INT,
    quantity INT,
    status VARCHAR(20) DEFAULT 'pending'
);

-- 创建订单时检查库存
DELIMITER $$
CREATE TRIGGER check_stock_before_order
BEFORE INSERT ON shop_orders
FOR EACH ROW
BEGIN
    DECLARE current_stock INT;
    
    SELECT stock INTO current_stock
    FROM shop_products
    WHERE id = NEW.product_id;
    
    IF current_stock < NEW.quantity THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = '库存不足';
    END IF;
END$$
DELIMITER ;

-- 订单确认后扣减库存
DELIMITER $$
CREATE TRIGGER update_stock_after_order
AFTER INSERT ON shop_orders
FOR EACH ROW
BEGIN
    UPDATE shop_products
    SET stock = stock - NEW.quantity,
        sold = sold + NEW.quantity
    WHERE id = NEW.product_id;
END$$
DELIMITER ;

-- 测试
INSERT INTO shop_products (name, stock) VALUES ('商品A', 100);
INSERT INTO shop_orders (product_id, quantity) VALUES (1, 10);
SELECT * FROM shop_products;
```

### 案例2：用户积分系统

```sql
-- 用户表
CREATE TABLE members (
    id INT PRIMARY KEY AUTO_INCREMENT,
    username VARCHAR(50),
    points INT DEFAULT 0,
    level VARCHAR(20) DEFAULT 'Bronze'
);

-- 积分记录表
CREATE TABLE point_records (
    id INT PRIMARY KEY AUTO_INCREMENT,
    user_id INT,
    points INT,
    reason VARCHAR(100),
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- 自动更新用户等级
DELIMITER $$
CREATE TRIGGER update_member_level
AFTER UPDATE ON members
FOR EACH ROW
BEGIN
    IF NEW.points != OLD.points THEN
        -- 根据积分更新等级
        IF NEW.points >= 10000 THEN
            UPDATE members SET level = 'Diamond' WHERE id = NEW.id;
        ELSEIF NEW.points >= 5000 THEN
            UPDATE members SET level = 'Gold' WHERE id = NEW.id;
        ELSEIF NEW.points >= 1000 THEN
            UPDATE members SET level = 'Silver' WHERE id = NEW.id;
        ELSE
            UPDATE members SET level = 'Bronze' WHERE id = NEW.id;
        END IF;
    END IF;
END$$
DELIMITER ;

-- 记录积分变化
DELIMITER $$
CREATE TRIGGER log_point_change
AFTER INSERT ON point_records
FOR EACH ROW
BEGIN
    UPDATE members
    SET points = points + NEW.points
    WHERE id = NEW.user_id;
END$$
DELIMITER ;
```

## 📝 本节总结
