## 网络架构设计

### 小型公司（50-100人）

```
互联网
  │
  ├─ 企业级路由器/防火墙
  │
  ├─ 核心交换机（三层）
  │   ├─ VLAN 10: 管理网络
  │   ├─ VLAN 20: 办公网络
  │   ├─ VLAN 30: 服务器网络
  │   ├─ VLAN 40: 访客网络
  │   └─ VLAN 50: 生产网络
  │
  ├─ 接入交换机（办公区）
  │   └─ 办公电脑/打印机
  │
  ├─ 无线AP控制器
  │   ├─ AP1（办公区）
  │   ├─ AP2（会议室）
  │   └─ AP3（访客区）
  │
  └─ 服务器区
      ├─ 域控服务器（AD）
      ├─ 文件服务器
      ├─ 邮件服务器
      ├─ 数据库服务器
      └─ 应用服务器
```

### VLAN配置

```bash
# 核心交换机VLAN配置

# 创建VLAN
vlan batch 10 20 30 40 50 60 70

# VLAN描述
vlan 10
  description Management-Network
vlan 20
  description Office-Network
vlan 30
  description Server-Network
vlan 40
  description Guest-Network
vlan 50
  description Development-Network
vlan 60
  description DMZ-Network
vlan 70
  description Monitor-Network

# 配置VLAN接口
interface Vlanif10
  ip address 172.16.10.1 255.255.255.0
  description Management-Gateway
  
interface Vlanif20
  ip address 172.16.20.1 255.255.252.0
  description Office-Gateway
  dhcp select relay
  dhcp relay server-ip 172.16.30.10
  
interface Vlanif30
  ip address 172.16.30.1 255.255.255.0
  description Server-Gateway
  
interface Vlanif40
  ip address 172.16.40.1 255.255.255.0
  description Guest-Gateway
  dhcp select relay
  dhcp relay server-ip 172.16.30.10
  
interface Vlanif50
  ip address 172.16.50.1 255.255.254.0
  description Development-Gateway
  
interface Vlanif60
  ip address 172.16.60.1 255.255.255.0
  description DMZ-Gateway
  
interface Vlanif70
  ip address 172.16.70.1 255.255.255.0
  description Monitor-Gateway

# 配置链路聚合（上联到防火墙）
interface Eth-Trunk1
  description Link-to-Firewall
  port link-type trunk
  port trunk allow-pass vlan all
  mode lacp
  
interface GigabitEthernet0/0/1
  eth-trunk 1
  
interface GigabitEthernet0/0/2
  eth-trunk 1
```

### 3. 访问控制列表（ACL）

```bash
# 核心交换机ACL配置

# 保护服务器网络
acl number 3000
  description Protect-Server-Network
  rule 5 permit ip source 172.16.10.0 0.0.0.255
  rule 10 permit ip source 172.16.20.0 0.0.3.255
  rule 15 permit ip source 172.16.60.0 0.0.0.255
  rule 20 deny ip source any
  
interface Vlanif30
  traffic-filter inbound acl 3000

# 隔离访客网络
acl number 3001
  description Isolate-Guest-Network
  rule 5 permit udp destination-port eq 53
  rule 10 permit tcp destination-port eq 80
  rule 15 permit tcp destination-port eq 443
  rule 20 deny ip destination 172.16.0.0 0.0.255.255
  rule 25 permit ip destination any
  
interface Vlanif40
  traffic-filter outbound acl 3001
```

### 2. 无线控制器配置

```bash
# 华为AC控制器配置

# 创建WLAN配置文件
wlan
  security-profile name office-security
    security wpa3 wpa3-enterprise authentication-method dot1x
    
  security-profile name guest-security
    security wpa2 psk pass-phrase GuestPassword2024!
    
  ssid-profile name office-ssid
    ssid Company-Office
    
  ssid-profile name guest-ssid
    ssid Company-Guest
    
  vap-profile name office-vap
    forward-mode tunnel
    service-vlan vlan-id 20
    ssid-profile office-ssid
    security-profile office-security
    
  vap-profile name guest-vap
    forward-mode tunnel
    service-vlan vlan-id 40
    ssid-profile guest-ssid
    security-profile guest-security
    user-isolate l2
    
# 配置AP组
ap-group name office-group
  vap-profile office-vap wlan 1 radio all
  vap-profile guest-vap wlan 2 radio all
  
# 配置AP
ap-id 1
  ap-name Office-AP-01
  ap-mac 00e0-fc12-3456
  ap-group office-group
```

### 2. 文件服务器

```bash
# Ubuntu文件服务器配置

# 安装Samba
apt install -y samba samba-common-bin

# 配置Samba
cat > /etc/samba/smb.conf << 'EOF'
[global]
    workgroup = COMPANY
    server string = File Server
    security = ads
    realm = COMPANY.COM
    encrypt passwords = yes
    
    # 日志配置
    log file = /var/log/samba/%m.log
    max log size = 50
    
    # 性能优化
    socket options = TCP_NODELAY IPTOS_LOWDELAY SO_RCVBUF=131072 SO_SNDBUF=131072
    read raw = yes
    write raw = yes
    
[public]
    path = /data/public
    browseable = yes
    writable = yes
    guest ok = no
    valid users = @"COMPANY\Domain Users"
    
[departments]
    path = /data/departments/%U
    browseable = yes
    writable = yes
    create mask = 0660
    directory mask = 0770
    valid users = @"COMPANY\Domain Users"
EOF

# 加入域
net ads join -U administrator

# 启动服务
systemctl restart smbd nmbd
systemctl enable smbd nmbd
```

### 2. 备份脚本

```bash
#!/bin/bash
# 自动备份脚本

BACKUP_DIR="/backup"
DATE=$(date +%Y%m%d_%H%M%S)
LOG_FILE="/var/log/backup.log"

echo "$(date): 开始备份" >> $LOG_FILE

# 备份文件服务器
rsync -avz --delete /data/ $BACKUP_DIR/fileserver/$DATE/

# 备份数据库
mysqldump --all-databases | gzip > $BACKUP_DIR/database/mysql_$DATE.sql.gz

# 备份配置文件
tar -czf $BACKUP_DIR/configs/configs_$DATE.tar.gz \
    /etc/samba \
    /etc/postfix \
    /etc/nginx

# 清理旧备份（保留30天）
find $BACKUP_DIR -type f -mtime +30 -delete

# 同步到云端
rclone sync $BACKUP_DIR remote:backup/

echo "$(date): 备份完成" >> $LOG_FILE
```
