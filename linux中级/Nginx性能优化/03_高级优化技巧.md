# 高级优化技巧

## HTTP/2 优化

### 启用 HTTP/2

```nginx
server {
    listen 443 ssl http2;
    listen [::]:443 ssl http2;
    server_name example.com;
    
    ssl_certificate /path/to/cert.crt;
    ssl_certificate_key /path/to/cert.key;
    
    # HTTP/2 配置
    http2_push_preload on;
    http2_max_concurrent_streams 128;
    http2_max_field_size 16k;
    http2_max_header_size 32k;
}
```

### HTTP/2 Server Push

```nginx
server {
    listen 443 ssl http2;
    server_name example.com;
    
    location / {
        root /data/www;
        index index.html;
        
        # 推送资源
        http2_push /css/style.css;
        http2_push /js/script.js;
    }
    
    # 或使用 Link 头
    location /page {
        add_header Link "</css/style.css>; rel=preload; as=style";
        add_header Link "</js/script.js>; rel=preload; as=script";
    }
}
```

### HTTP/2 vs HTTP/1.1

```
HTTP/1.1：
- 每个请求一个连接
- 队头阻塞
- 明文头部

HTTP/2：
- 多路复用（一个连接多个请求）
- 头部压缩（HPACK）
- Server Push
- 二进制协议

性能提升：20-50%
```

## CDN 加速

### CDN 工作原理

```
用户请求
    ↓
CDN 节点（就近）
    ↓
源站（Nginx）

优势：
- 减少延迟
- 降低带宽成本
- 提高可用性
- 防御 DDoS
```

### 配置 CDN 回源

```nginx
server {
    listen 80;
    server_name origin.example.com;
    
    # 只允许 CDN 回源
    allow 1.2.3.0/24;    # CDN IP 段
    deny all;
    
    # 获取真实 IP
    set_real_ip_from 1.2.3.0/24;
    real_ip_header X-Forwarded-For;
    
    location / {
        root /data/www;
        
        # 缓存控制
        expires 7d;
        add_header Cache-Control "public, max-age=604800";
    }
}
```

### 静态资源分离

```nginx
# 主站
server {
    listen 80;
    server_name www.example.com;
    
    location / {
        root /data/www;
    }
    
    # 静态资源使用 CDN
    location ~* \.(jpg|jpeg|png|gif|css|js)$ {
        return 301 https://cdn.example.com$request_uri;
    }
}

# CDN 源站
server {
    listen 80;
    server_name cdn.example.com;
    
    location / {
        root /data/static;
        expires 30d;
        add_header Cache-Control "public, immutable";
    }
}
```

## 负载均衡优化

### 连接复用

```nginx
upstream backend {
    server 192.168.1.101:8080;
    server 192.168.1.102:8080;
    
    # 保持连接
    keepalive 32;
    keepalive_timeout 60s;
    keepalive_requests 100;
}

server {
    location / {
        proxy_pass http://backend;
        
        # 使用 HTTP/1.1
        proxy_http_version 1.1;
        
        # 清除 Connection 头
        proxy_set_header Connection "";
    }
}
```

### 健康检查

```nginx
upstream backend {
    server 192.168.1.101:8080 max_fails=3 fail_timeout=30s;
    server 192.168.1.102:8080 max_fails=3 fail_timeout=30s;
    server 192.168.1.103:8080 backup;
    
    # 被动健康检查
    # 3 次失败后标记不可用，30 秒后重试
}

# 主动健康检查（需要 Nginx Plus 或第三方模块）
```

### 会话保持优化


## 缓存策略优化

### 多级缓存

```
浏览器缓存
    ↓
CDN 缓存
    ↓
Nginx 缓存
    ↓
应用缓存（Redis）
    ↓
数据库
```

### 缓存预热

```bash
#!/bin/bash
# cache_warmup.sh

URLS=(
    "http://example.com/"
    "http://example.com/page1"
    "http://example.com/page2"
)

for url in "${URLS[@]}"; do
    echo "预热: $url"
    curl -s -o /dev/null "$url"
done

echo "缓存预热完成"
```

### 缓存清理

```nginx
# 配置清理接口
location ~ /purge(/.*) {
    allow 127.0.0.1;
    allow 192.168.1.0/24;
    deny all;
    
    proxy_cache_purge my_cache $scheme$proxy_host$1;
}

# 使用
# curl -X PURGE http://example.com/purge/page1
```

### 缓存分片

```nginx
http {
    # 按 URL 分片
    proxy_cache_path /cache/shard1 levels=1:2 keys_zone=cache1:10m;
    proxy_cache_path /cache/shard2 levels=1:2 keys_zone=cache2:10m;
    
    map $request_uri $cache_zone {
        ~^/api/  cache1;
        default  cache2;
    }
    
    server {
        location / {
            proxy_cache $cache_zone;
            proxy_pass http://backend;
        }
    }
}
```

## 安全优化

### 防 DDoS

```nginx
# 限制请求速率
limit_req_zone $binary_remote_addr zone=req_limit:10m rate=10r/s;
limit_req_zone $server_name zone=server_limit:10m rate=100r/s;

# 限制连接数
limit_conn_zone $binary_remote_addr zone=conn_limit:10m;
limit_conn_zone $server_name zone=server_conn_limit:10m;

server {
    # 应用限制
    limit_req zone=req_limit burst=20 nodelay;
    limit_req zone=server_limit burst=100 nodelay;
    limit_conn conn_limit 10;
    limit_conn server_conn_limit 1000;
    
    # 慢速攻击防护
    client_body_timeout 10s;
    client_header_timeout 10s;
    send_timeout 10s;
}
```

### 防爬虫

```nginx
# 方法1：User-Agent 过滤
map $http_user_agent $bad_bot {
    default 0;
    ~*(scrapy|curl|wget|python|java) 1;
}

server {
    if ($bad_bot) {
        return 403;
    }
}

# 方法2：请求频率限制
limit_req_zone $binary_remote_addr zone=bot_limit:10m rate=1r/s;

location / {
    limit_req zone=bot_limit burst=5;
}

# 方法3：验证码
location / {
    if ($http_user_agent ~* bot) {
        return 302 /captcha;
    }
}
```

### 防盗链

```nginx
location ~* \.(jpg|jpeg|png|gif)$ {
    valid_referers none blocked example.com *.example.com;
    
    if ($invalid_referer) {
        return 403;
    }
    
    expires 30d;
}
```

## 动静分离

### 架构设计

```
                Nginx（前端）
                    │
        ┌───────────┴───────────┐
        │                       │
    静态资源              动态请求
        │                       │
    本地文件              后端应用
  /data/static          (PHP/Java/Node.js)
```

### 配置示例

```nginx
server {
    listen 80;
    server_name www.example.com;
    
    # 静态资源
    location ~* \.(jpg|jpeg|png|gif|ico|css|js|svg|woff|woff2)$ {
        root /data/static;
        expires 30d;
        add_header Cache-Control "public, immutable";
        access_log off;
    }
    
    # 动态请求
    location / {
        proxy_pass http://backend;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
    }
}
```

## 微服务架构优化

### API 网关

```nginx
# 用户服务
upstream user_service {
    least_conn;
    server 192.168.1.101:8001;
    server 192.168.1.102:8001;
    keepalive 32;
}

# 订单服务
upstream order_service {
    least_conn;
    server 192.168.1.101:8002;
    server 192.168.1.102:8002;
    keepalive 32;
}

# 支付服务
upstream payment_service {
    least_conn;
    server 192.168.1.101:8003;
    server 192.168.1.102:8003;
    keepalive 32;
}

server {
    listen 80;
    server_name api.example.com;
    
    # 限流
    limit_req_zone $binary_remote_addr zone=api_limit:10m rate=100r/s;
    limit_req zone=api_limit burst=200 nodelay;
    
    # 用户服务
    location /api/users/ {
        proxy_pass http://user_service/;
        proxy_http_version 1.1;
        proxy_set_header Connection "";
    }
    
    # 订单服务
    location /api/orders/ {
        proxy_pass http://order_service/;
        proxy_http_version 1.1;
        proxy_set_header Connection "";
    }
    
    # 支付服务
    location /api/payment/ {
        proxy_pass http://payment_service/;
        proxy_http_version 1.1;
        proxy_set_header Connection "";
    }
}
```

### 服务熔断

```nginx
# 使用 error_page 实现简单熔断
upstream backend {
    server 192.168.1.101:8080 max_fails=3 fail_timeout=30s;
    server 192.168.1.102:8080 max_fails=3 fail_timeout=30s;
}

server {
    location / {
        proxy_pass http://backend;
        proxy_next_upstream error timeout http_500 http_502 http_503;
        
        error_page 502 503 504 = @fallback;
    }
    
    location @fallback {
        return 200 '{"error":"Service temporarily unavailable"}';
        add_header Content-Type application/json;
    }
}
```

## WebSocket 优化

```nginx
map $http_upgrade $connection_upgrade {
    default upgrade;
    '' close;
}

upstream websocket {
    server 192.168.1.101:8080;
    server 192.168.1.102:8080;
    
    # WebSocket 需要较长超时
    keepalive 32;
}

server {
    listen 80;
    server_name ws.example.com;
    
    location / {
        proxy_pass http://websocket;
        
        # WebSocket 必需配置
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection $connection_upgrade;
        
        # 超时设置
        proxy_connect_timeout 7d;
        proxy_send_timeout 7d;
        proxy_read_timeout 7d;
        
        # 其他配置
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
    }
}
```

## 容器化部署优化

### Docker 优化

```dockerfile
# Dockerfile
FROM nginx:alpine

# 复制优化后的配置
COPY nginx.conf /etc/nginx/nginx.conf
COPY conf.d/ /etc/nginx/conf.d/

# 复制网站文件
COPY html/ /usr/share/nginx/html/

# 健康检查
HEALTHCHECK --interval=30s --timeout=3s \
  CMD wget --quiet --tries=1 --spider http://localhost/health || exit 1

EXPOSE 80

CMD ["nginx", "-g", "daemon off;"]
```
